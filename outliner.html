    li { margin: .25rem 0; cursor: pointer; }

    li > span { display: inline-block; padding: 2px 4px; border-radius: 6px; }

  </div>


<script>

(async function () {

  const res = await fetch('./slm.opml', { cache: 'no-store' });

  const text = await res.text();

  const xml = new DOMParser().parseFromString(text, 'text/xml');


  // OPML: <body><outline text="..."><outline .../></outline></body>

  const rootOutlines = Array.from(xml.querySelectorAll('opml > body > outline'));


  // Build a lightweight tree with stable ids

  let idCounter = 0;

  function build(node, parentId=null) {

    const id = String(++idCounter);

    const text = node.getAttribute('text') || '';

    const note = node.getAttribute('_note') || node.getAttribute('note') || '';

    const children = Array.from(node.children).filter(n => n.tagName === 'outline').map(ch => build(ch, id));

    return { id, parentId, text, note, children };

  }

  const forest = rootOutlines.map(n => build(n));


  // index by id

  const byId = new Map();

  (function index(nodes){

    for (const n of nodes) { byId.set(n.id, n); index(n.children); }

  })(forest);


  function getPath(nodeId) {

    const path = [];

    let cur = byId.get(nodeId);

    while (cur) { path.unshift(cur); cur = cur.parentId ? byId.get(cur.parentId) : null; }

    return path;

  }


  function render(nodeId=null) {

    const container = document.getElementById('view');

    const crumbs = document.getElementById('crumbs');

    container.innerHTML = '';

    crumbs.innerHTML = '';


    let nodesToShow, path=[];

    if (!nodeId) {

      nodesToShow = forest;

      crumbs.textContent = 'Home';

    } else {

      const node = byId.get(nodeId);

      path = getPath(nodeId);

      // breadcrumbs

      const parts = ['<a href="#"></a>'];

      crumbs.innerHTML = path.map((p, i) => {

        const isLast = i === path.length - 1;

        return isLast

          ? `<strong>${escapeHtml(p.text)}</strong>`

          : `<a href="#node=${p.id}">${escapeHtml(p.text)}</a> / `;

      }).join('');

      nodesToShow = node.children;

    }


    container.appendChild(renderList(nodesToShow));

  }


  function renderList(nodes) {

    const ul = document.createElement('ul');

    for (const n of nodes) {

      const li = document.createElement('li');

      const label = document.createElement('span');

      label.textContent = n.text || '(empty)';

      label.onclick = () => { location.hash = `node=${n.id}`; };

      li.appendChild(label);


      if (n.note) {

        const note = document.createElement('span');

        note.className = 'note';

        note.textContent = '— ' + n.note;

        li.appendChild(note);

      }

      ul.appendChild(li);

    }

    return ul;

  }


  function escapeHtml(s){ return (s||'').replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])); }


  function onHash() {

    const m = location.hash.match(/node=(\d+)/);

    render(m ? m[1] : null);

  }

  window.addEventListener('hashchange', onHash);

  onHash();

})();

</script>
